\title{On the Kaplan--Meier Plateau in Recurrent Gap-Time Data}
\author{Jacob Spear}
\date{\today}

\begin{document}
\maketitle

\begin{abstract}
We show that the Kaplan--Meier estimator applied to inter-visit
gap times converges to $1/\bar{v}$, where $\bar{v}$ is the mean
number of visits per customer, regardless of the underlying churn
dynamics. This makes it unsuitable for estimating cure fractions
in recurrent-event settings.
\end{abstract}
